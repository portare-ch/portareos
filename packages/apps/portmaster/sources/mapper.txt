#!/bin/bash

. /etc/profile.d/001-functions
. /etc/profile.d/100-gamecontroller-functions

# -- Config & Setup --
# Destination file
if [[ -z "$1" ]] && [[ -z "$2" ]]; then
  echo "Usage: mapper.txt [gamecontrollerdb.txt] [controlfolder]"
  exit -1
fi

CONTROLLER_DB="$1"
controlfolder="$2"

#Scrub the CONTROLLER_DB file
echo "" > ${CONTROLLER_DB}

# Where the emulationstation configuration file is
ES_CONFIG="/tmp/es_input.cfg"

#Set layout via Emulation Station
ACTIVE_GAME=""
ACTIVE_PLATFORM=""
GAME="${ACTIVE_GAME##*/}"
PLATFORM="${ACTIVE_PLATFORM##*/}"
CONTROLLER_LAYOUT=$(get_setting controller_layout "${PLATFORM}" "${GAME}")

#Default to nintendo if no value assigned
if [ ! -n "${CONTROLLER_LAYOUT}" ]; then
  CONTROLLER_LAYOUT="nintendo"
fi

scriptdir="controller_layout"

if [ ! -d "${controlfolder}/${scriptdir}" ]; then
    mkdir -p "${controlfolder}/${scriptdir}"
fi

echo -e "# Xbox Layout\nABUT=\"b\"\nBBUT=\"a\"\nXBUT=\"y\"\nYBUT=\"x\"" >  "${controlfolder}/${scriptdir}/xbox_layout.txt"
echo -e "# Nintendo Layout\nABUT=\"a\"\nBBUT=\"b\"\nXBUT=\"x\"\nYBUT=\"y\"" >  "${controlfolder}/${scriptdir}/nintendo_layout.txt"
if [[ ! -e "${controlfolder}/${scriptdir}/custom_layout.txt" ]]; then
  echo -e "# Custom Layout\nABUT=\"a\"\nBBUT=\"b\"\nXBUT=\"x\"\nYBUT=\"y\"" >  "${controlfolder}/${scriptdir}/custom_layout.txt"
fi

source "${controlfolder}/${scriptdir}/${CONTROLLER_LAYOUT}_layout.txt"

# Call function to create controller db from ES input config file for only connected devices
create_controller_db ${CONTROLLER_LAYOUT} ${ES_CONFIG} ${CONTROLLER_DB}

# The Nova's own pad. InputPlumber presents it as a DualSense Edge, and
# SDL's built-in map for that names the bottom button a and the right one
# b, which on this device are printed B and A. The function above would
# have written a map with the names following the labels, but it works
# from EmulationStation's input config, which this image no longer has,
# so nothing was written and PortMaster took SDL's names: A opened, B
# went back, each on the wrong button. This is SDL's own map for the pad
# with the four face buttons put through the chosen layout: nintendo,
# the default, keeps the printed letters; xbox keeps SDL's positions.
# Later lines win in SDL, so this goes after everything else.
printf '0300e0274c050000f20d000000006800,Retroid Pocket Nova,%s:b1,%s:b0,%s:b3,%s:b2,back:b4,dpdown:h0.4,dpleft:h0.8,dpright:h0.2,dpup:h0.1,guide:b5,leftshoulder:b9,leftstick:b7,lefttrigger:a4,leftx:a0,lefty:a1,rightshoulder:b10,rightstick:b8,righttrigger:a5,rightx:a2,righty:a3,start:b6,paddle1:b16,paddle2:b15,paddle3:b14,paddle4:b13,platform:Linux,\n' \
  "${ABUT:-a}" "${BBUT:-b}" "${XBUT:-x}" "${YBUT:-y}" >> "${CONTROLLER_DB}"

#Reset file for next run
sed -i '/^ACTIVE_GAME=/c\ACTIVE_GAME=""' /storage/.config/PortMaster/mapper.txt
sed -i '/^ACTIVE_PLATFORM=/c\ACTIVE_PLATFORM=""' /storage/.config/PortMaster/mapper.txt
